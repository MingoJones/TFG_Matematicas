\begin{frame}{¿Cómo verificar la equidistribución?}

\begin{alertblock}{Problema}
La definición exige comprobar que
\[
\lim_{N\to\infty}\frac{A_N((a,b),x)}{N}=b-a
\]
para todo intervalo $(a,b)\subset\mathbb T$.
\end{alertblock}

\vspace{0.3cm}
\pause

\begin{exampleblock}{Objetivo}
Buscar criterios más prácticos para demostrar
que una sucesión es equidistribuida.
\end{exampleblock}


\end{frame}


\subsection{Caracterización funcional}
% Caracterización funcional
\begin{frame}{Caracterización funcional}

\begin{block}{Teorema}
Las siguientes afirmaciones son equivalentes:
\begin{enumerate}
\item $(x_n)$ es equidistribuida módulo $1$.
\item Para toda función Riemann integrable $f:\mathbb T\to\mathbb R$,
\[
\lim_{N\to\infty}\frac{1}{N}\sum_{n=1}^{N}f(x_n)=\int_0^1 f(x)\,dx.
\]
\end{enumerate}
\end{block}

\vspace{0.2cm}
\pause

\begin{block}{Corolario}
Este teorema también es válido para funciones continuas.
\end{block}

\end{frame}


\subsection{Criterio de Weyl}
% Criterio de Weyl
\begin{frame}{Criterio de Weyl}

\begin{block}{Criterio de Weyl}
Las siguientes afirmaciones son equivalentes:

\begin{enumerate}
\item $(x_n)$ es equidistribuida módulo $1$.
\item Para todo $k\in\mathbb Z\setminus\{0\}$,

\[
\lim_{N\to\infty} \frac{1}{N}\sum_{n=1}^{N}e^{2\pi i k x_n}=0
\]
\end{enumerate}
\end{block}

\vspace{0.2cm}
\pause

\begin{exampleblock}{Ventaja}
La equidistribución puede comprobarse estudiando
únicamente las sumas de exponenciales complejas.
\end{exampleblock}

\end{frame}

% Aplicación del criterio de Weyl
\begin{frame}{Aplicación del criterio de Weyl}

\begin{block}{Resultado}
Si $\alpha\notin\mathbb Q$, entonces $(n\alpha)_{n\geq1}$
es equidistribuida módulo $1$.
\end{block}

\vspace{0.1cm}

Por el criterio de Weyl, basta demostrar que
\[
\lim_{N\to\infty}\frac{1}{N}\sum_{n=1}^{N}e^{2\pi i k n\alpha}=0
\qquad (k \neq 0).
\]

\pause

La suma es geométrica y satisface
\[
\left|\frac{1}{N}\sum_{n=1}^{N} e^{2\pi i k n\alpha}\right|
\le\frac{2}{N|e^{2\pi i k\alpha}-1|}.
\]

Por tanto, el limite es $0$ y $(n\alpha)$ es equidistribuida.
\end{frame}


% EJEMPLO sin n
\begin{frame}{Aplicación del criterio de Weyl}

\begin{block}{Resultado}
La sucesión $(\sin n)_{n\ge1}$ no es equidistribuida módulo $1$.
\end{block}

\vspace{0.1cm}
\pause

Observamos que $\frac{1}{2\pi}\notin\mathbb Q,$
por lo que $\left(\frac{n}{2\pi}\right)$ es equidistribuida módulo $1$.
Aplicando la caracterización funcional a
\[
f(x)=e^{2\pi i\sin(2\pi x)},
\]
obtenemos
\[
\lim_{N\to\infty} \frac1N \sum_{n=1}^{N} e^{2\pi i\sin n}
= \int_0^1 e^{2\pi i\sin(2\pi x)}\,dx = J_0(2\pi) \approx 0.22 \neq 0.
\]

\begin{alertblock}{Conclusión}
El criterio de Weyl falla para $k=1$, por lo tanto, $(\sin n)$ no es equidistribuida módulo $1$.
\end{alertblock}

\end{frame}
