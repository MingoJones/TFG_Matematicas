\subsection{Números reales módulo 1}
\begin{frame}{Números reales módulo 1}
      
\begin{block}{Parte fraccionaria}
Sea $x\in \mathbb{R}$, definimos la parte fraccionaria de $x$ como
\[
\partfrac{x} = x - \floor{x}.
\]
Observamos que $\partfrac{x} \in [0,1)$ para todo $x \in \mathbb{R}$.
\end{block}

\pause

\begin{block}{Módulo 1}
Consideramos la relación de equivalencia en $\mathbb{R}$ 
dada por
\[
x \sim y \iff x-y \in \mathbb{Z}.
\]
Dos números $x$ e $y$ son congruentes módulo $1$ si 
coinciden en su parte fraccionaria.
\end{block}

\end{frame}


\begin{frame}{Números reales módulo 1}
El espacio creado por la relación de equivalencia se denota por 
$\mathbb{T} \simeq \mathbb{R}/\mathbb{Z}$.

\pause

\begin{exampleblock}{Observación}
Las funciones en $\mathbb{T}$ se identifican con las funciones 
$1$-periódicas en $\mathbb{R}$:
\[
f(x+1) = f(x) \qquad x \in \mathbb{R}.
\]
Por lo que $f(x) = f(\partfrac{x})$ para todo $x \in \mathbb{R}$.
\end{exampleblock}

\end{frame}


\subsection{Equidistribución de sucesiones}
\begin{frame}{Equidistribución de sucesiones}

\begin{block}{Definición}
Sea $x=(x_n)_{n\geq1}$ una sucesión de números reales. 
Decimos que $(x_n)$ es equidistribuida módulo $1$ si para cada
intervalo $(a,b) \subset \mathbb{T}$ se cumple que
\[
\label{def:equidistribucion}
\lim_{N\to\infty} \frac{A_N((a,b), x)}{N} = b-a
\]
donde 
\[
A_N((a,b), x) = 
\card \left\{1\le n\le N : \partfrac{x_n} \in (a,b)\right\}.
\]
\end{block}

\pause

\begin{exampleblock}{Observación}
Si consideramos $\mathbf{1}_{(a,b)}$ en $\mathbb{T}$, la
definición se puede reescribir como
\[
\lim_{N\to\infty} \frac{1}{N} \sum_{n=1}^N \mathbf{1}_{(a,b)}(x_n) =
\int_0^1 \mathbf{1}_{(a,b)}(x)\ dx = 
b-a.
\]
\end{exampleblock}

\end{frame}


\begin{frame}{Propiedades y ejemplos}

\begin{block}{Propiedades}
\begin{itemize}
\item Sea $\alpha \in \mathbb{R}$ y $(x_n)$ equidistribuida, entonces $(x_n+\alpha)$ también lo es.
Es~decir, la equidistribución es invariante por traslaciones.
\vspace{0.2cm}
\item Modificar un número finito de términos no altera la equidistribución.
\end{itemize}
\end{block}

\vspace{0.2cm}
\pause

\begin{exampleblock}{Ejemplos}
La siguiente sucesión es equidistribuida:
\[
0, \ 
0, \ \frac{1}{2}, \ 
0, \ \frac{1}{3}, \ \frac{2}{3}, \ 
0, \ \frac{1}{4}, \ \frac{2}{4}, \ \frac{3}{4}, \ 
0, \ \dots 
\]
\end{exampleblock}

\end{frame}


\subsection{Sucesión $n\alpha$}
\begin{frame}{Sucesión $(n\alpha)$}

\begin{block}{Proposición}

\begin{itemize}
\item Si $\alpha\in\mathbb Q$, la sucesión $(\partfrac{n\alpha})$ es periódica y tiene un número finito de términos distintos.
\item Si $\alpha\notin\mathbb Q$, la sucesión $(\partfrac{n\alpha})$ tiene infinitos términos distintos.
\end{itemize}

\end{block}

\begin{frame}{Sucesión $(n\alpha)$}

\begin{block}{Teorema de Weyl}
Si $\alpha\notin\mathbb Q$, la sucesión $(n\alpha)$ es equidistribuida módulo $1$.
\end{block}

\vspace{0.2cm}
\pause

\begin{block}{Lema}
Sea $f \in C(\mathbb{T})$ y sea $\alpha \notin \mathbb{Q}$. Entonces, 
\[
      \frac{1}{N} \sum_{n=1}^{N} f(n\alpha) \longrightarrow \int_0^1 f(x)\ dx
      \qquad \text{cuando } N \to \infty .
\] 
\end{block}

\end{frame}

\begin{frame}{Sucesión $(n\alpha)$}
\only<1,3->{
\textbf{Dem.} Sea $\varepsilon > 0$ y sean $f_\varepsilon^-, f_\varepsilon^+ \in C(\mathbb{T})$ tales que
\[
\mathbf{1}_{(a+\varepsilon, b-\varepsilon)} \leq 
f_\varepsilon^- \leq 
\mathbf{1}_{(a,b)} \leq 
f_\varepsilon^+ \leq
\mathbf{1}_{(a-\varepsilon, b+\varepsilon)}
\]
\onslide<3->{
Sea $S_N = \sum_{n=1}^N \mathbf{1}_{(a,b)}(n\alpha)$. Entonces,
\[
\frac{1}{N} \sum_{n=1}^{N} f_\varepsilon^-(n\alpha) \leq
\frac{S_N}{N} \leq
\frac{1}{N} \sum_{n=1}^{N} f_\varepsilon^+(n\alpha).
\]
}
\onslide<4->{
Haciendo $N \to \infty$ y usando el lema,
\[
b-a-2\varepsilon \le
\int_0^1 f_\varepsilon^- \le
\liminf_{N\to\infty}\frac{S_N}{N} \le
\limsup_{N\to\infty}\frac{S_N}{N} \le
\int_0^1 f_\varepsilon^+ \le
b-a+2\varepsilon,
\]

\[
\lim_{N\to\infty}\frac{S_N}{N} = b-a.
\]
}
}
