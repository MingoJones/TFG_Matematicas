\subsection{Desigualdad de Van der Corput}
\begin{frame}{Desigualdad de Van der Corput}

\begin{block}{Desigualdad de Van der Corput}
Sean $H,N\in\mathbb N$ con $1\le H\le N$. Entonces

\[
\left|\sum_{n=1}^{N} e^{2\pi i f(n)} \right|^2
\le
4 \, \frac{N}{H} \sum_{h=0}^{H-1} \left|\sum_{n=1}^{N-h}e^{2\pi i(f(n+h)-f(n))}\right|.
\]
\end{block}

\vspace{0.2cm}
\pause

\begin{exampleblock}{Idea}
La suma exponencial asociada a $f(n)$ puede controlarse
mediante sumas exponenciales asociadas a las diferencias
\[
f(n+h)-f(n).
\]
\end{exampleblock}

\end{frame}


\subsection{Teorema de las diferencias de Van der Corput}
\begin{frame}{Sucesión $(n^2\alpha)$}

\begin{block}{Resultado}
La sucesión $(n^2\alpha)$ con $\alpha\notin\mathbb Q$ es equidistribuida módulo $1$.
\end{block}

\pause 

Sea
\[
f(n)=kn^2\alpha \quad \text{con } k \in \mathbb{Z} \setminus \{0\}.
\]
Entonces
\[
f(n+h)-f(n)=k(2hn\alpha+h^2\alpha).
\]

\pause

Sabemos que $(n\alpha)$ es equidistribuida módulo $1$. 
Entonces 
\[
(2hn\alpha+h^2\alpha)
\] 
es equidistribuida módulo $1$ para cada $h \neq 0$.

\end{frame}

\begin{frame}{Sucesión $(n^2\alpha)$}

Aplicando la desigualdad de Van der Corput y el criterio
de Weyl, se obtiene,

\[
\limsup_{N\to\infty}
\left|
\frac1N
\sum_{n=1}^{N}
e^{2\pi ikn^2\alpha}
\right|^2
\le
\frac4H.
\]

\vspace{0.3cm}
\pause


Haciendo $H\to\infty$ se concluye que

\[
\lim_{N\to\infty}
\frac1N
\sum_{n=1}^{N}
e^{2\pi ikn^2\alpha}
=0.
\]

\begin{block}{Conclusión}
Por el criterio de Weyl, $(n^2\alpha)$ es equidistribuida módulo $1$.
\end{block}

\end{frame}


\begin{frame}{Teorema de las diferencias de Van der Corput}
\begin{block}{Teorema de las diferencias de Van der Corput}
Sea $(x_n)$ una sucesión de números reales.
Si para todo entero positivo $h$, $(x_{n+h}-x_n)$ es equidistribuida 
módulo $1$, entonces
\[
(x_n)
\]
también es equidistribuida módulo $1$.
\end{block}

\vspace{0.2cm}
\pause

\begin{exampleblock}{Importancia}
El resultado permite sustituir una sucesión complicada
por sus sucesiones de diferencias, las cuales pueden ser más sencillas de estudiar.
\end{exampleblock}

\end{frame}


\subsection{Aplicaciones del teorema de las diferencias}
\begin{frame}{Polinomios con coeficientes irracionales}

\begin{block}{Teorema}
Sea $p(x)=c_mx^m+\cdots+c_1x+c_0$ un polinomio con coeficientes reales.

Si al menos uno de los coeficientes
$c_1,\ldots,c_m$ es irracional, entonces
\[
(p(n))
\]
es equidistribuida módulo $1$.
\end{block}

\pause

\begin{demostracion}
\textbf{Caso base:}
\[
c_1\notin\mathbb Q,
\qquad
c_2,\ldots,c_m\in\mathbb Q.
\]
Tomando $D$ como un denominador común de $c_2, \ldots, c_m$ y escribiendo $n = Dl + d$, 
las sumas exponenciales se reducen al estudio de 
\[
(c_1Dl)_{l\geq0}
\]
Como $(c_1Dl)$ es equidistribuida módulo $1$, $(p(n))$ también lo es.

\end{demostracion}
\end{frame}

\begin{frame}{Polinomios con coeficientes irracionales}

\textbf{Caso general:}

\vspace{0.2cm}

Por inducción, sea $s$ el mayor grado tal que $c_s\notin\mathbb Q$.
Consideramos las diferencias
\[
p_h(x)=p(x+h)-p(x).
\]

\vspace{0.1cm}
\pause

Al desarrollar,
\[
p_h(x)
=
\sum_{i=0}^{m-1}\alpha_i x^i.
\]

El coeficiente irracional dominante pasa de grado
$s$ a grado $s-1$.

\vspace{0.1cm}

Por hipótesis de inducción, $(p_h(n))$ es equidistribuida 
para todo $h\ge 1$.

\vspace{0.1cm}
\pause

Aplicando el teorema de las diferencias de
Van der Corput,
\[
(p(n))
\]
es equidistribuida módulo $1$.

\end{frame}

% Teorema de Fejer
\begin{frame}{Teorema de Fejér Generalizado}
\begin{block}{Teorema}
Sea $g:[1,\infty)\to\mathbb R$ una función $k$ veces derivable con
$g^{(k)}\in C([1,\infty))$ y $g^{(k)}$ monótona para $x$ 
suficientemente grande. Si se satisfacen

\[
\lim_{x\to\infty}g^{(k)}(x)=0
\quad \text{y} \quad
\lim_{x\to\infty}x\,|g^{(k)}(x)|=\infty,
\]

entonces $(g(n))_{n\geq1}$ es equidistribuida módulo $1$.
\end{block}

\vspace{0.2cm}
\pause 

\begin{demostracion}
Por inducción sobre $k$. El caso $k=1$ es el teorema de Fejér.

\vspace{0.15cm}

Sea $k>1$, comprobamos que las diferencias $g(n+h)-g(n)$ cumplen
las hipótesis del teorema para $k-1$. 

\vspace{0.15cm}

Aplicando el teorema de las diferencias de Van der Corput, 
se concluye que $(g(n))$ es equidistribuida módulo $1$.
\end{demostracion}

\end{frame}

\begin{frame}{Teorema de Fejér Generalizado}

\begin{exampleblock}{Ejemplo 1}
La sucesión $(an^\sigma)_{n\geq1}$ con $a\neq 0$, $\sigma > 0,\, \sigma\notin\mathbb{Z}$ es equidistribuida módulo~$1$.
\end{exampleblock}
\pause
\begin{exampleblock}{Ejemplo 2}
La sucesión $(a n^\sigma \log^\tau n)_{n\geq2}$  con $a\neq 0$,  $\tau\in\mathbb R$, $\sigma > 0,\, \sigma\notin\mathbb{Z}$  es equidistribuida módulo $1$.
\end{exampleblock}
\pause
\begin{exampleblock}{Ejemplo 3}
La sucesión $(a n^k \log^\tau n)_{n\geq2}$  con $a\neq 0$, $\tau<0$, $ k\geq1,\, k\in\mathbb Z$  es equidistribuida módulo $1$.
\end{exampleblock}

\end{frame}
